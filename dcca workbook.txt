---------------- MainActivity.java ----------------
package com.example.myapp;

import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    // STEP 1: declare one variable for each widget
    EditText etName;
    Button btnSubmit;
    TextView tvResult;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);   // STEP 2: show the XML screen

        // STEP 3: connect each variable to its XML id
        etName = findViewById(R.id.etName);
        btnSubmit = findViewById(R.id.btnSubmit);
        tvResult = findViewById(R.id.tvResult);

        // STEP 4: what happens when button is clicked
        btnSubmit.setOnClickListener(v -> {

            // STEP 5: read what the user typed
            String name = etName.getText().toString().trim();

            // STEP 6a: validation - empty?
            if (name.isEmpty()) {
                etName.requestFocus();                    // put cursor there
                etName.setError("FIELD CANNOT BE EMPTY"); // red error message
                return;                                   // STOP here
            }

            // STEP 6b: validation - only letters and spaces allowed
            if (!name.matches("[a-zA-Z ]+")) {
                etName.requestFocus();
                etName.setError("ENTER ONLY ALPHABETICAL CHARACTERS");
                return;
            }

            // STEP 7: do the work
            tvResult.setText("Hello " + name);

            // STEP 8: small popup message
            Toast.makeText(this, "Submitted!", Toast.LENGTH_SHORT).show();
        });
    }
}

REMEMBER:
- getText() gives "Editable", so we add .toString() to make it a String.
- .trim() removes extra spaces at start/end.
- Without "return;" the code keeps going even when there is an error.
- Toast needs 3 things: (this, "message", Toast.LENGTH_SHORT) and .show()

2. INTENT: OPEN SECOND SCREEN + SEND DATA
=====================================================================
Idea: Intent = a parcel. putExtra = put a labelled item in the parcel.
      getStringExtra = open the parcel on the other side using SAME label.

XML IDs NEEDED:
  activity_main.xml   : EditText etName, EditText etDept, Button btnNext
  activity_second.xml : TextView tvShow

MANIFEST (inside <application>):
  <activity android:name=".SecondActivity" />

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    EditText etName, etDept;
    Button btnNext;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etName = findViewById(R.id.etName);
        etDept = findViewById(R.id.etDept);
        btnNext = findViewById(R.id.btnNext);

        btnNext.setOnClickListener(v -> {
            String name = etName.getText().toString().trim();
            String dept = etDept.getText().toString().trim();

            if (name.isEmpty()) {
                etName.requestFocus();
                etName.setError("FIELD CANNOT BE EMPTY");
                return;
            }
            if (dept.isEmpty()) {
                etDept.requestFocus();
                etDept.setError("FIELD CANNOT BE EMPTY");
                return;
            }

            // make the parcel: FROM this screen TO SecondActivity
            Intent intent = new Intent(MainActivity.this, SecondActivity.class);
            intent.putExtra("name", name);   // label "name"
            intent.putExtra("dept", dept);   // label "dept"
            startActivity(intent);           // send it / open screen
        });
    }
}

---------------- SecondActivity.java ----------------
package com.example.myapp;

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.os.Bundle;
import android.widget.Button;
import android.widget.Toast;

import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    Button btnSimple, btnThree, btnList;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        btnSimple = findViewById(R.id.btnSimple);
        btnThree = findViewById(R.id.btnThree);
        btnList = findViewById(R.id.btnList);

        // KIND 1: Yes / No dialog
        btnSimple.setOnClickListener(v -> {
            new AlertDialog.Builder(MainActivity.this)
                    .setTitle("Exit")
                    .setMessage("Are you sure you want to exit?")
                    .setIcon(android.R.drawable.ic_dialog_alert)
                    .setCancelable(false)   // can't close by tapping outside
                    .setPositiveButton("Yes", (dialog, which) -> finish())
                    .setNegativeButton("No", (dialog, which) -> dialog.dismiss())
                    .show();
        });

        // KIND 2: Three buttons (max 3: Positive, Negative, Neutral)
        btnThree.setOnClickListener(v -> {
            new AlertDialog.Builder(MainActivity.this)
                    .setTitle("Save Changes")
                    .setMessage("Do you want to save?")
                    .setPositiveButton("Save", (dialog, which) ->
                            Toast.makeText(this, "Saved", Toast.LENGTH_SHORT).show())
                    .setNegativeButton("Don't Save", (dialog, which) ->
                            Toast.makeText(this, "Not saved", Toast.LENGTH_SHORT).show())
                    .setNeutralButton("Cancel", (dialog, which) -> dialog.dismiss())
                    .show();
        });

        // KIND 3: List of choices
        btnList.setOnClickListener(v -> {
            String[] colors = {"Red", "Green", "Blue"};
            new AlertDialog.Builder(MainActivity.this)
                    .setTitle("Pick a colour")
                    .setItems(colors, (dialog, which) ->
                            Toast.makeText(this, "You chose " + colors[which],
                                    Toast.LENGTH_SHORT).show())
                    .show();
        });
    }
}

4. SQLITE: DBHelper CLASS (the notebook maker)
=====================================================================
STORY:
  Database = a notebook inside the phone.
  Table    = one page with columns:   id | name | marks
  DBHelper = the shopkeeper who makes the notebook:
     onCreate  -> first time: draw the page (CREATE TABLE)
     onUpgrade -> design changed: tear the page out, draw again
  4 jobs only: INSERT (write row), SELECT (read rows),
               UPDATE (change row), DELETE (rub out row)

  ContentValues = a blank FORM you fill:  cv.put("column", value)
  Cursor        = your FINGER moving down the rows: moveToNext()
  getWritableDatabase() = open notebook with a PEN   (insert/update/delete)
  getReadableDatabase() = open notebook to READ      (select)
  "id=?" + new String[]{id}  = "the row where id is ___"

Make a new Java class: right-click package > New > Java Class > DBHelper

---------------- DBHelper.java ----------------
package com.example.myapp;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

public class DBHelper extends SQLiteOpenHelper {

    // notebook name and version
    private static final String DB_NAME = "StudentDB";
    private static final int DB_VERSION = 1;

    // page (table) name and column names
    private static final String TABLE = "student";
    private static final String COL_ID = "id";
    private static final String COL_NAME = "name";
    private static final String COL_MARKS = "marks";

    public DBHelper(Context context) {
        super(context, DB_NAME, null, DB_VERSION);
    }

    // runs ONLY ONCE - when the database is created first time
    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL("CREATE TABLE " + TABLE + " ("
                + COL_ID + " INTEGER PRIMARY KEY, "
                + COL_NAME + " TEXT, "
                + COL_MARKS + " INTEGER)");
    }

    // runs when DB_VERSION is increased
    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
        db.execSQL("DROP TABLE IF EXISTS " + TABLE);
        onCreate(db);
    }

    // ---------------- INSERT: write a new row ----------------
    public boolean insertData(String id, String name, String marks) {
        SQLiteDatabase db = this.getWritableDatabase();   // pen mode
        ContentValues cv = new ContentValues();           // blank form
        cv.put(COL_ID, id);                               // fill boxes
        cv.put(COL_NAME, name);
        cv.put(COL_MARKS, marks);
        long result = db.insert(TABLE, null, cv);         // stick form in
        db.close();
        return result != -1;                              // -1 means failed
    }

    // ---------------- SELECT: read all rows ----------------
    public Cursor getAllData() {
        SQLiteDatabase db = this.getReadableDatabase();   // read mode
        return db.rawQuery("SELECT * FROM " + TABLE, null);
    }

    // ---------------- UPDATE: change a row ----------------
    public boolean updateData(String id, String name, String marks) {
        SQLiteDatabase db = this.getWritableDatabase();
        ContentValues cv = new ContentValues();           // form with new values
        cv.put(COL_NAME, name);
        cv.put(COL_MARKS, marks);
        int rows = db.update(TABLE, cv, COL_ID + "=?", new String[]{id});
        db.close();
        return rows > 0;                                  // 0 = id not found
    }

    // ---------------- DELETE: rub out a row ----------------
    public boolean deleteData(String id) {
        SQLiteDatabase db = this.getWritableDatabase();
        int rows = db.delete(TABLE, COL_ID + "=?", new String[]{id});  // no form!
        db.close();
        return rows > 0;
    }
}

COMPARE THE 4 LINES:
  db.insert(TABLE, null, cv)                          -> form, no "which row"
  db.update(TABLE, cv, "id=?", new String[]{id})      -> form + which row
  db.delete(TABLE, "id=?", new String[]{id})          -> which row only
  db.rawQuery("SELECT * FROM student", null)          -> read everything

5. SQLITE: MainActivity WITH INSERT / VIEW / UPDATE / DELETE  -> Exp 7
=====================================================================
XML IDs NEEDED (activity_main.xml):
  EditText etId      (inputType="number")
  EditText etName
  EditText etMarks   (inputType="number")
  Button   btnInsert, btnView, btnUpdate, btnDelete
  TextView tvData

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.database.Cursor;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    EditText etId, etName, etMarks;
    Button btnInsert, btnView, btnUpdate, btnDelete;
    TextView tvData;
    DBHelper db;    // our notebook maker

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        db = new DBHelper(this);    // create the helper

        etId = findViewById(R.id.etId);
        etName = findViewById(R.id.etName);
        etMarks = findViewById(R.id.etMarks);
        btnInsert = findViewById(R.id.btnInsert);
        btnView = findViewById(R.id.btnView);
        btnUpdate = findViewById(R.id.btnUpdate);
        btnDelete = findViewById(R.id.btnDelete);
        tvData = findViewById(R.id.tvData);

        // ---------- INSERT button ----------
        btnInsert.setOnClickListener(v -> {
            String id = etId.getText().toString().trim();
            String name = etName.getText().toString().trim();
            String marks = etMarks.getText().toString().trim();

            if (id.isEmpty() || name.isEmpty() || marks.isEmpty()) {
                Toast.makeText(this, "Fill all fields", Toast.LENGTH_SHORT).show();
                return;
            }

            boolean ok = db.insertData(id, name, marks);
            if (ok) {
                Toast.makeText(this, "Data Inserted", Toast.LENGTH_SHORT).show();
            } else {
                Toast.makeText(this, "Insert Failed (id may already exist)",
                        Toast.LENGTH_SHORT).show();
            }
        });

        // ---------- VIEW (SELECT) button ----------
        btnView.setOnClickListener(v -> {
            Cursor c = db.getAllData();

            if (c.getCount() == 0) {          // no rows at all
                tvData.setText("No data found");
                c.close();
                return;
            }

            StringBuilder sb = new StringBuilder();   // to join all rows
            while (c.moveToNext()) {                  // finger goes row by row
                sb.append("ID: ").append(c.getString(0))      // column 0
                  .append("  Name: ").append(c.getString(1))  // column 1
                  .append("  Marks: ").append(c.getString(2)) // column 2
                  .append("\n");
            }
            c.close();
            tvData.setText(sb.toString());
        });

        // ---------- UPDATE button ----------
        btnUpdate.setOnClickListener(v -> {
            String id = etId.getText().toString().trim();
            String name = etName.getText().toString().trim();
            String marks = etMarks.getText().toString().trim();

            if (id.isEmpty()) {
                etId.requestFocus();
                etId.setError("ENTER ID TO UPDATE");
                return;
            }

            boolean ok = db.updateData(id, name, marks);
            Toast.makeText(this, ok ? "Data Updated" : "ID not found",
                    Toast.LENGTH_SHORT).show();
        });

        // ---------- DELETE button ----------
        btnDelete.setOnClickListener(v -> {
            String id = etId.getText().toString().trim();

            if (id.isEmpty()) {
                etId.requestFocus();
                etId.setError("ENTER ID TO DELETE");
                return;
            }

            boolean ok = db.deleteData(id);
            Toast.makeText(this, ok ? "Data Deleted" : "ID not found",
                    Toast.LENGTH_SHORT).show();
        });
    }
}

NOTE: ok ? "A" : "B"  means  "if ok is true use A, otherwise use B".

6. DATEPICKERDIALOG + TIMEPICKERDIALOG
=====================================================================
Idea: Calendar.getInstance() = today's date/time (where the picker starts).
      The picker pops up, user picks, and your code receives the values.
      !! Month comes 0-based (January = 0), so always add 1.

XML IDs NEEDED:
  TextView tvDate, Button btnDate, TextView tvTime, Button btnTime

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.app.DatePickerDialog;
import android.app.TimePickerDialog;
import android.os.Bundle;
import android.widget.Button;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

import java.util.Calendar;

public class MainActivity extends AppCompatActivity {

    TextView tvDate, tvTime;
    Button btnDate, btnTime;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        tvDate = findViewById(R.id.tvDate);
        tvTime = findViewById(R.id.tvTime);
        btnDate = findViewById(R.id.btnDate);
        btnTime = findViewById(R.id.btnTime);

        // ---------- DATE PICKER ----------
        btnDate.setOnClickListener(v -> {
            Calendar c = Calendar.getInstance();          // today
            int year = c.get(Calendar.YEAR);
            int month = c.get(Calendar.MONTH);
            int day = c.get(Calendar.DAY_OF_MONTH);

            DatePickerDialog dpd = new DatePickerDialog(
                    MainActivity.this,
                    (view, y, m, d) -> {
                        // this runs AFTER the user picks a date
                        tvDate.setText(d + "/" + (m + 1) + "/" + y);   // m + 1 !!
                    },
                    year, month, day);                    // starting date

            dpd.getDatePicker().setMaxDate(System.currentTimeMillis()); // no future date
            dpd.show();
        });

        // ---------- TIME PICKER ----------
        btnTime.setOnClickListener(v -> {
            Calendar c = Calendar.getInstance();
            int hour = c.get(Calendar.HOUR_OF_DAY);
            int minute = c.get(Calendar.MINUTE);

            TimePickerDialog tpd = new TimePickerDialog(
                    MainActivity.this,
                    (view, h, min) -> {
                        // this runs AFTER the user picks a time
                        tvTime.setText(String.format("%02d:%02d", h, min));
                    },
                    hour, minute,
                    true);                 // true = 24 hour, false = AM/PM
            tpd.show();
        });
    }
}

SPINNER-STYLE DATE PICKER (like Exp 4) - add a theme as 2nd parameter:
    new DatePickerDialog(MainActivity.this,
            android.R.style.Theme_Holo_Light_Dialog_NoActionBar,
            (view, y, m, d) -> { ... }, year, month, day).show();

7. FILE HANDLING: WRITE TO FILE + READ FROM FILE   -> Exp 6
=====================================================================
Idea: Your app has a private folder in the phone. We write a text file
      there and read it back.
      openFileOutput = open file to WRITE    openFileInput = open file to READ
      MODE_PRIVATE = overwrite old content   MODE_APPEND = add at the end

XML IDs NEEDED:
  EditText etData, Button btnWrite, Button btnRead, TextView tvResult

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.content.Context;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.io.BufferedReader;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.InputStreamReader;

public class MainActivity extends AppCompatActivity {

    EditText etData;
    Button btnWrite, btnRead;
    TextView tvResult;

    String FILE_NAME = "mydata.txt";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etData = findViewById(R.id.etData);
        btnWrite = findViewById(R.id.btnWrite);
        btnRead = findViewById(R.id.btnRead);
        tvResult = findViewById(R.id.tvResult);

        btnWrite.setOnClickListener(v -> writeFile());
        btnRead.setOnClickListener(v -> readFile());
    }

    // ---------- WRITE ----------
    private void writeFile() {
        String data = etData.getText().toString().trim();

        if (data.isEmpty()) {
            etData.requestFocus();
            etData.setError("FIELD CANNOT BE EMPTY");
            return;
        }

        try {
            FileOutputStream fos = openFileOutput(FILE_NAME, Context.MODE_APPEND);
            fos.write((data + "\n").getBytes());   // text -> bytes -> file
            fos.close();
            Toast.makeText(this, "Data written to file", Toast.LENGTH_SHORT).show();
            etData.setText("");
        } catch (IOException e) {
            Toast.makeText(this, "Error: " + e.getMessage(), Toast.LENGTH_SHORT).show();
        }
    }

    // ---------- READ ----------
    private void readFile() {
        try {
            FileInputStream fis = openFileInput(FILE_NAME);
            BufferedReader br = new BufferedReader(new InputStreamReader(fis));

            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) {   // read line by line until end
                sb.append(line).append("\n");
            }
            br.close();

            tvResult.setText(sb.toString());
            Toast.makeText(this, "Data read from file", Toast.LENGTH_SHORT).show();
        } catch (IOException e) {
            tvResult.setText("File not found. Write something first.");
        }
    }
}

WHY try/catch? File work can fail (file missing etc.), and Java FORCES
you to handle that with try { } catch (IOException e) { }.

9. LISTVIEW + SIMPLEADAPTER (text + image)   -> Exp 4
=====================================================================
Idea: each ROW = one HashMap  {"name" -> "Potato", "image" -> picture}
      all rows = ArrayList of HashMaps
      from = the HashMap labels       {"name", "image"}
      to   = where they go (view ids) {R.id.tvName, R.id.ivImage}
      SimpleAdapter(context, data, resource, from, to)  <- learn these 5!

XML IDs NEEDED:
  activity_main.xml : TextView tvDate, ListView listView
  list_item.xml     : TextView tvName, ImageView ivImage   (design of ONE row)
DRAWABLE: potato, onion, tomato, carrot images in res/drawable (small letters)

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.app.DatePickerDialog;
import android.os.Bundle;
import android.widget.ListView;
import android.widget.SimpleAdapter;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.util.ArrayList;
import java.util.Calendar;
import java.util.HashMap;

public class MainActivity extends AppCompatActivity {

    ListView listView;
    TextView tvDate;

    String[] names = {"Potato", "Onion", "Tomato", "Carrot"};
    int[] images = {R.drawable.potato, R.drawable.onion,
                    R.drawable.tomato, R.drawable.carrot};

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        listView = findViewById(R.id.listView);
        tvDate = findViewById(R.id.tvDate);

        // 1. build the data: one HashMap per row
        ArrayList<HashMap<String, Object>> data = new ArrayList<>();
        for (int i = 0; i < names.length; i++) {
            HashMap<String, Object> row = new HashMap<>();
            row.put("name", names[i]);
            row.put("image", images[i]);
            data.add(row);
        }

        // 2. from (labels) and to (view ids) - SAME ORDER
        String[] from = {"name", "image"};
        int[] to = {R.id.tvName, R.id.ivImage};

        // 3. make the adapter: context, data, resource, from, to
        SimpleAdapter adapter = new SimpleAdapter(this, data,
                R.layout.list_item, from, to);

        // 4. attach to ListView
        listView.setAdapter(adapter);

        // row click
        listView.setOnItemClickListener((parent, view, position, id) ->
                Toast.makeText(MainActivity.this,
                        names[position] + " is available", Toast.LENGTH_SHORT).show());

        // tap the date to change it (spinner style)
        Calendar c = Calendar.getInstance();
        tvDate.setText(c.get(Calendar.DAY_OF_MONTH) + "/"
                + (c.get(Calendar.MONTH) + 1) + "/" + c.get(Calendar.YEAR));

        tvDate.setOnClickListener(v -> {
            new DatePickerDialog(MainActivity.this,
                    android.R.style.Theme_Holo_Light_Dialog_NoActionBar,
                    (view, y, m, d) -> tvDate.setText(d + "/" + (m + 1) + "/" + y),
                    c.get(Calendar.YEAR), c.get(Calendar.MONTH),
                    c.get(Calendar.DAY_OF_MONTH)).show();
        });
    }
}

10. WEBVIEW: STATIC HTML   -> Exp 5 (screen 2)

Idea: WebView = a mini browser box inside your screen.

---------------- StaticActivity.java ----------------
package com.example.myapp;

import android.os.Bundle;
import android.webkit.WebView;

import androidx.appcompat.app.AppCompatActivity;

public class StaticActivity extends AppCompatActivity {

    WebView webview;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_static);

        webview = findViewById(R.id.webview);

        String html = "<html><body>"
                + "<h1>Welcome to NMIMS</h1>"
                + "<p>It's a Static Web HTML Content</p>"
                + "<ul><li>Academics</li><li>Campus</li></ul>"
                + "</body></html>";

        webview.loadData(html, "text/html", "UTF-8");
    }
}

11. WEBVIEW: REMOTE WEBSITE + VALIDATION + ALERTDIALOG   -> Exp 5
=====================================================================
Idea: loadUrl("https://...") = load a real website from the internet.
      Needs INTERNET permission, otherwise blank screen.

XML IDs NEEDED:
  activity_main.xml   : Button btnStatic, Button btnRemote
  activity_remote.xml : EditText etUrl, Button btnDisplay, WebView webview

MANIFEST:
  <uses-permission android:name="android.permission.INTERNET" />   (outside <application>)
  <activity android:name=".StaticActivity" />
  <activity android:name=".RemoteActivity" />

---------------- MainActivity.java (first screen with 2 buttons) ----------------
package com.example.myapp;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    Button btnStatic, btnRemote;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        btnStatic = findViewById(R.id.btnStatic);
        btnRemote = findViewById(R.id.btnRemote);

        btnStatic.setOnClickListener(v ->
                startActivity(new Intent(MainActivity.this, StaticActivity.class)));

        btnRemote.setOnClickListener(v ->
                startActivity(new Intent(MainActivity.this, RemoteActivity.class)));
    }
}

---------------- RemoteActivity.java ----------------
package com.example.myapp;

import android.os.Bundle;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.widget.Button;
import android.widget.EditText;

import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;

public class RemoteActivity extends AppCompatActivity {

    EditText etUrl;
    Button btnDisplay;
    WebView webview;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_remote);

        etUrl = findViewById(R.id.etUrl);
        btnDisplay = findViewById(R.id.btnDisplay);
        webview = findViewById(R.id.webview);

        webview.setWebViewClient(new WebViewClient());     // open inside app, not Chrome
        webview.getSettings().setJavaScriptEnabled(true);  // most sites need this

        btnDisplay.setOnClickListener(v -> {
            String url = etUrl.getText().toString().trim();

            // VALIDATION 1: empty
            if (url.isEmpty()) {
                etUrl.requestFocus();
                etUrl.setError("FIELD CANNOT BE EMPTY");
                return;
            }

            // VALIDATION 2: must look like a website (something.something)
            if (!url.matches("^(https?://)?(www\\.)?[a-zA-Z0-9-]+(\\.[a-zA-Z]{2,})+(/.*)?$")) {
                etUrl.requestFocus();
                etUrl.setError("ENTER A VALID WEBSITE (e.g. www.google.com)");
                return;
            }

            // add https:// if the user didn't type it
            if (!url.startsWith("http://") && !url.startsWith("https://")) {
                url = "https://" + url;
            }

            String finalUrl = url;   // lambdas below need a variable that doesn't change

            // ALERT DIALOG before opening
            new AlertDialog.Builder(RemoteActivity.this)
                    .setTitle("Open Website")
                    .setMessage("Do you want to open " + finalUrl + " ?")
                    .setIcon(android.R.drawable.ic_dialog_info)
                    .setPositiveButton("Open", (dialog, which) -> webview.loadUrl(finalUrl))
                    .setNegativeButton("Cancel", (dialog, which) -> dialog.dismiss())
                    .setNeutralButton("Clear", (dialog, which) -> etUrl.setText(""))
                    .show();
        });
    }
}

12. GOOGLE MAPS: MARKER AT LATITUDE / LONGITUDE   -> Exp 8
=====================================================================
Idea: The map loads slowly, so we ask "tell me when ready"
      (getMapAsync). Android then calls onMapReady() and gives us the map.
      Only inside onMapReady can we add markers.

SETUP:
 (a) build.gradle.kts (Module :app) -> dependencies { } -> add, then Sync Now:
       implementation("com.google.android.gms:play-services-maps:18.1.0")
 (b) MANIFEST:
       <uses-permission android:name="android.permission.INTERNET" />
       inside <application>:
       <meta-data
           android:name="com.google.android.geo.API_KEY"
           android:value="YOUR_API_KEY" />
 (c) XML (activity_main.xml):
       EditText etLat, EditText etLng, Button btnLocate, and the map:
       <fragment
           android:id="@+id/map"
           android:name="com.google.android.gms.maps.SupportMapFragment"
           android:layout_width="match_parent"
           android:layout_height="match_parent" />

---------------- MainActivity.java ----------------
package com.example.myapp;

import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import com.google.android.gms.maps.CameraUpdateFactory;
import com.google.android.gms.maps.GoogleMap;
import com.google.android.gms.maps.OnMapReadyCallback;
import com.google.android.gms.maps.SupportMapFragment;
import com.google.android.gms.maps.model.LatLng;
import com.google.android.gms.maps.model.MarkerOptions;

public class MainActivity extends AppCompatActivity implements OnMapReadyCallback {

    GoogleMap mMap;
    EditText etLat, etLng;
    Button btnLocate;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etLat = findViewById(R.id.etLat);
        etLng = findViewById(R.id.etLng);
        btnLocate = findViewById(R.id.btnLocate);

        // find the map box and ask: "call onMapReady when loaded"
        SupportMapFragment mapFragment = (SupportMapFragment)
                getSupportFragmentManager().findFragmentById(R.id.map);
        mapFragment.getMapAsync(this);

        btnLocate.setOnClickListener(v -> {
            String latStr = etLat.getText().toString().trim();
            String lngStr = etLng.getText().toString().trim();

            if (latStr.isEmpty() || lngStr.isEmpty()) {
                Toast.makeText(this, "Enter latitude and longitude",
                        Toast.LENGTH_SHORT).show();
                return;
            }

            double lat = Double.parseDouble(latStr);   // text -> number
            double lng = Double.parseDouble(lngStr);

            if (lat < -90 || lat > 90 || lng < -180 || lng > 180) {
                Toast.makeText(this, "Invalid coordinates", Toast.LENGTH_SHORT).show();
                return;
            }

            showPlace(lat, lng, "Selected Location");
        });
    }

    // Android calls this when the map is ready
    @Override
    public void onMapReady(GoogleMap googleMap) {
        mMap = googleMap;
        mMap.getUiSettings().setZoomControlsEnabled(true);   // + and - buttons
        showPlace(22.7186, 75.8547, "Rajwada Palace, Indore");  // default place
    }

    private void showPlace(double lat, double lng, String title) {
        if (mMap == null) return;                 // map not ready yet
        LatLng place = new LatLng(lat, lng);      // the point
        mMap.clear();                             // remove old marker
        mMap.addMarker(new MarkerOptions()        // put red pin
                .position(place)
                .title(title));
        mMap.animateCamera(CameraUpdateFactory.newLatLngZoom(place, 15f)); // zoom there
    }
}

ZOOM: 1 = world, 5 = continent, 10 = city, 15 = streets, 20 = buildings
API KEY: console.cloud.google.com -> new project -> enable "Maps SDK for
Android" -> Credentials -> Create credentials -> API key -> paste in manifest.
SHA-1: Gradle panel -> Execute task -> "gradle signingReport"

14. COMMON MISTAKES (crash or lose marks)
=====================================================================
- Activity not in manifest -> crash when you open it
- INTERNET permission missing -> WebView / Map is blank
- Forgot .show() on Toast or AlertDialog -> nothing appears
- Toast missing Toast.LENGTH_SHORT -> error
- DatePicker month not + 1 -> shows wrong month
- Small/capital letter mistakes: findViewById, setOnClickListener, EditText
- R.id name in Java different from @+id name in XML -> crash
- Forgot "return;" after setError -> code continues with bad data
- putExtra label different from getStringExtra label -> you get null
- SQL missing spaces: "CREATE TABLE" + TABLE -> "CREATE TABLEstudent"
- db.delete needs 3 things: (table, "id=?", new String[]{id})
- Image names in drawable must be small letters, no spaces
